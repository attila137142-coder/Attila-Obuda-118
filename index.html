const express = require("express"); const app = express(); const fetch = require("node-fetch"); app.use(express.json()); let stops = []; let stopTimes = []; let vehicles = []; // ---------- START ---------- app.listen(3000, () => { console.log("🚀 FUTÁR PRO MAX backend fut: 3000"); }); // ---------- GTFS LOAD (SIMPLIFIED) ---------- const fs = require("fs"); function loadGTFS(){ stops = fs.readFileSync("stops.txt","utf8") .split("\n").slice(1) .map(r=>{ let c=r.split(","); return {id:c[0],name:c[2],lat:+c[4],lng:+c[5]}; }); stopTimes = fs.readFileSync("stop_times.txt","utf8") .split("\n").slice(1) .map(r=>{ let c=r.split(","); return {trip:c[0],time:c[2],stop:c[3]}; }); console.log("GTFS betöltve"); } loadGTFS(); // ---------- STOPS API ---------- app.get("/stops",(req,res)=>{ res.json(stops); }); // ---------- DEPARTURES ---------- app.get("/departures/:stopId",(req,res)=>{ let now = new Date().toTimeString().slice(0,5); let deps = stopTimes .filter(t=>t.stop === req.params.stopId && t.time > now) .slice(0,10); res.json(deps); }); // ---------- REALTIME VEHICLES ---------- async function updateVehicles(){ // 🔴 MOCK (később GTFS-RT ide jön) vehicles = stops.slice(0,20).map((s,i)=>({ id:i, lat:s.lat + Math.random()*0.001, lng:s.lng + Math.random()*0.001, route:"BKK-" + (5+i), delay: Math.floor(Math.random()*15) })); } setInterval(updateVehicles,5000); app.get("/vehicles",(req,res)=>{ res.json(vehicles); });
