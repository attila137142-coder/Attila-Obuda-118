# =========================================================# 🟢 FULL GTFS EXPORT ENGINE V1# BKK / GTFS → PDF + XLSX EXPORTER# ---------------------------------------------------------# FUNKCIÓK:# ✔ GTFS ZIP letöltés# ✔ stops.txt feldolgozás# ✔ Budapest szűrés# ✔ Pest megye szűrés# ✔ ABC sorrend# ✔ koordináták 6 tizedesig# ✔ XLSX export# ✔ PDF export# =========================================================import osimport zipfileimport requestsimport pandas as pdfrom io import BytesIOfrom reportlab.platypus import ( SimpleDocTemplate, Table, TableStyle, Paragraph, Spacer, PageBreak)from reportlab.lib import colorsfrom reportlab.lib.styles import getSampleStyleSheetfrom reportlab.lib.pagesizes import landscape, A4# =========================================================# CONFIG# =========================================================GTFS_URL = "https://go.bkk.hu/api/static/v1/public-gtfs/budapest_gtfs.zip"OUTPUT_XLSX = "GTFS_EXPORT.xlsx"OUTPUT_PDF = "GTFS_EXPORT.pdf"# =========================================================# DOWNLOAD GTFS# =========================================================print("🟢 GTFS letöltés...")r = requests.get(GTFS_URL)if r.status_code != 200: raise Exception("❌ GTFS letöltés sikertelen!")print("🟢 ZIP megnyitás...")z = zipfile.ZipFile(BytesIO(r.content))# =========================================================# STOPS.TXT BETÖLTÉS# =========================================================print("🟢 stops.txt feldolgozás...")with z.open("stops.txt") as f: stops = pd.read_csv(f)# =========================================================# BIZTONSÁGI CHECK# =========================================================required = [ "stop_id", "stop_name", "stop_lat", "stop_lon"]for col in required: if col not in stops.columns: raise Exception(f"❌ Hiányzó oszlop: {col}")# =========================================================# KOORDINÁTA FORMÁZÁS# =========================================================stops["stop_lat"] = stops["stop_lat"].round(6)stops["stop_lon"] = stops["stop_lon"].round(6)# =========================================================# BUDAPEST SZŰRÉS# =========================================================print("🟢 Budapest szűrés...")budapest = stops[ stops["stop_name"] .astype(str) .str.contains("Budapest|M |tér|utca|út|krt", case=False, na=False)].copy()# fallbackif len(budapest) < 1000: budapest = stops.copy()# =========================================================# PEST MEGYE SZŰRÉS (egyszerű közelítés)# =========================================================print("🟢 Pest megye szűrés...")pest_keywords = [ "Szentendre", "Vác", "Dunakeszi", "Göd", "Érd", "Budaörs", "Pomáz", "Veresegyház", "Fót", "Monor", "Cegléd", "Nagykőrös", "Aszód", "Gödöllő", "Pilis", "Vecsés",]pattern = "|".join(pest_keywords)pest = stops[ stops["stop_name"] .astype(str) .str.contains(pattern, case=False, na=False)].copy()# =========================================================# ABC SORREND# =========================================================budapest = budapest.sort_values("stop_name")pest = pest.sort_values("stop_name")# =========================================================# OSZLOPOK# =========================================================columns = [ "stop_name", "stop_id", "stop_lat", "stop_lon"]budapest = budapest[columns]pest = pest[columns]# =========================================================# XLSX EXPORT# =========================================================print("🟢 XLSX generálás...")with pd.ExcelWriter(OUTPUT_XLSX, engine="openpyxl") as writer: budapest.to_excel( writer, sheet_name="Budapest_BKK", index=False ) pest.to_excel( writer, sheet_name="Pest_Megye", index=False )print(f"✅ XLSX kész: {OUTPUT_XLSX}")# =========================================================# PDF EXPORT# =========================================================print("🟢 PDF generálás...")doc = SimpleDocTemplate( OUTPUT_PDF, pagesize=landscape(A4))styles = getSampleStyleSheet()elements = []# ---------------------------------------------------------# TITLE# ---------------------------------------------------------elements.append( Paragraph("GTFS EXPORT ENGINE", styles["Title"]))elements.append(Spacer(1, 12))# =========================================================# PDF TABLE FUNCTION# =========================================================def add_table(title, df): elements.append( Paragraph(title, styles["Heading2"]) ) data = [list(df.columns)] for _, row in df.iterrows(): data.append([ str(row["stop_name"]), str(row["stop_id"]), str(row["stop_lat"]), str(row["stop_lon"]) ]) t = Table(data) t.setStyle(TableStyle([ ("BACKGROUND", (0, 0), (-1, 0), colors.lightgrey), ("GRID", (0, 0), (-1, -1), 0.5, colors.black), ("FONTNAME", (0, 0), (-1, 0), "Helvetica-Bold"), ("FONTSIZE", (0, 0), (-1, -1), 7), ("BOTTOMPADDING", (0, 0), (-1, 0), 8), ])) elements.append(t) elements.append(PageBreak())# =========================================================# TABLES# =========================================================add_table("Budapest BKK Megállók", budapest.head(1000))add_table("Pest Megye Megállók", pest.head(1000))# =========================================================# BUILD PDF# =========================================================doc.build(elements)print(f"✅ PDF kész: {OUTPUT_PDF}")# =========================================================# DONE# =========================================================print()print("🚀 GTFS EXPORT ENGINE KÉSZ")print("📄 PDF:", OUTPUT_PDF)print("📊 XLSX:", OUTPUT_XLSX)print("🟢 Feldolgozás befejezve")
