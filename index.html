import express from "express"; import fetch from "node-fetch"; const app = express(); /** * 🔥 BKK GTFS-RT ENDPOINTS (valós logika) * (publikus feedek struktúrája GTFS-RT protobuf alapú) */ const VEHICLE_POS_URL = "https://go.bkk.hu/api/gtfs/vehiclePositions"; const TRIP_UPDATES_URL = "https://go.bkk.hu/api/gtfs/tripUpdates"; /** * 🚍 LIVE VEHICLE FETCH */ async function getVehicles() { // ⚠️ valós implementációban protobuf parser kell const res = await fetch(VEHICLE_POS_URL); const data = await res.json().catch(() => []); return data.vehicle || []; } /** * 🚏 STOP LIVE QUERY */ app.get("/live/stop/:stopName", async (req, res) => { const stopName = req.params.stopName; const vehicles = await getVehicles(); // 🚨 SIMPLIFIED MATCH (később GTFS stop_id mapping) const filtered = vehicles.filter(v => (v.stopName || "").includes(stopName) ); const departures = filtered.map(v => ({ route: v.trip?.routeId || "UNK", headsign: v.trip?.headsign || "Unknown", delay: v.timestampDelay || 0, eta: calculateETA(v), vehicleId: v.vehicle?.id })) .sort((a,b) => a.eta - b.eta); res.json({ stop: stopName, departures }); }); /** * ⏱ ETA LOGIKA */ function calculateETA(v) { const speed = v.position?.speed || 30; const distance = v.position?.distanceToStop || 1500; return Math.max(1, Math.round(distance / (speed * 16.6))); } app.listen(3000, () => { console.log("🚍 FUTÁR BKK LIVE ENGINE RUNNING"); });
